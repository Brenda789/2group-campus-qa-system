•  请求生命周期：前端 →  Controller → Service → Mapper → MySQL →  逐层返回

统一响应格式  Result<T>

所有接口返回同一种结构，前端处理更简单

public class Result<T> {
    private int code;         // 200 成功  / 400 参数错  / 401 未授权  / 500 错误
    private String message;
    private T data;
    private long timestamp = System.currentTimeMillis();

@Data

    public static <T> Result<T> success(T data) {
        Result<T> r = new Result<>();
        r.code = 200; r.message = "success"; r.data = data;
        return r;
    }
    public static <T> Result<T> error(int code, String msg) {
        Result<T> r = new Result<>();
        r.code = code; r.message = msg;
        return r;
    }
}

返回示例： { "code":200, "message":"success", "data":{...} }

全局异常处理

@RestControllerAdvice  统一捕获异常，避免每个接口都  try-catch

public class GlobalExceptionHandler {

@RestControllerAdvice

    @ExceptionHandler(BizException.class)          //  自定义业务异常
    public Result<Void> biz(BizException e) {
        return Result.error(e.getCode(), e.getMessage());
    }

    @ExceptionHandler(MethodArgumentNotValidException.class)  //  参数校验失败
    public Result<Void> valid(MethodArgumentNotValidException e) {
        String msg = e.getBindingResult().getFieldError().getDefaultMessage();
        return Result.error(400, msg);
    }

    @ExceptionHandler(Exception.class)             //  兜底
    public Result<Void> all(Exception e) {
        return Result.error(500, " 服务器错误 : " + e.getMessage());
    }
}

    User u = new User();

public class JwtUtil {
    private static final String SECRET = "campus-qa-secret-key-2026";
    private static final long EXPIRE = 24 * 60 * 60 * 1000L;  // 24  小时

@Component

    public String generate(Long userId, String role) {       //  签发
        return Jwts.builder()
            .setSubject(String.valueOf(userId))
            .claim("role", role)
            .setExpiration(new Date(System.currentTimeMillis() + EXPIRE))
            .signWith(Keys.hmacShaKeyFor(SECRET.getBytes()))
            .compact();
    }
    public Claims parse(String token) {                       //  解析 / 校验
        return Jwts.parserBuilder()
            .setSigningKey(Keys.hmacShaKeyFor(SECRET.getBytes()))
            .build().parseClaimsJws(token).getBody();
    }
}

public Result<Map<String,Object>> login(@RequestBody LoginDTO dto) {
    User u = userService.lambdaQuery()
             .eq(User::getUsername, dto.getUsername()).one();
    // BCrypt  比对：明文  vs  密文
    if (u == null || !new BCryptPasswordEncoder()
            .matches(dto.getPassword(), u.getPassword()))
        throw new BizException(401, " 用户名或密码错误 ");
    if (u.getStatus() == 0) throw new BizException(403, " 账号已被禁用 ");

@PostMapping("/login")

    String token = jwtUtil.generate(u.getId(), u.getRole());
    Map<String,Object> res = new HashMap<>();
    res.put("token", token);
    res.put("username", u.getUsername());
    res.put("role", u.getRole());
    return Result.success(res);
}

登录拦截器：统一校验  Token

在拦截器里校验  Token ，未登录直接拦截

public class JwtInterceptor implements HandlerInterceptor {
    @Resource private JwtUtil jwtUtil;

@Component

    @Override
    public boolean preHandle(HttpServletRequest req,
                             HttpServletResponse resp, Object handler) {
        String auth = req.getHeader("Authorization");
        if (auth == null || !auth.startsWith("Bearer "))
            throw new BizException(401, " 未登录 ");
        try {
            Claims c = jwtUtil.parse(auth.substring(7));
            req.setAttribute("userId", c.getSubject());
            req.setAttribute("role", c.get("role"));
            return true;
        } catch (Exception e) {
            throw new BizException(401, "Token  无效或已过期 ");
        }
    }
}

// src/services/request.ts

import axios from "axios";
const request = axios.create({ baseURL: "/api", timeout: 10000 });

//  请求拦截：自动携带  token
request.interceptors.request.use(cfg => {
  const token = localStorage.getItem("token");
  if (token) cfg.headers.Authorization = `Bearer ${token}`;
  return cfg;
});
//  响应拦截：统一解包与错误处理
request.interceptors.response.use(
  res => {
    const { code, message, data } = res.data;
    if (code !== 200) { return Promise.reject(message); }
    return data;
  },
  err => {
    if (err.response?.status === 401) location.href = "/login";  //  未登录跳转
    return Promise.reject(err);
  }
);
export default request;

import { useEffect, useState } from "react";
import { Table, Input, Tag } from "antd";
import request from "../services/request";

export default function UserList() {
  const [data, setData] = useState<any[]>([]);
  const [total, setTotal] = useState(0);
  const load = async (p = 1, keyword = "") => {
    const res = await request.get("/user/list",
                    { params: { page: p, size: 10, keyword } });
    setData(res.records); setTotal(res.total);
  };
  useEffect(() => { load(); }, []);
  const columns = [
    { title: " 用户名 ", dataIndex: "username" },
    { title: " 邮箱 ",   dataIndex: "email" },
    { title: " 角色 ",   dataIndex: "role", render: (r) => <Tag>{r}</Tag> },
    { title: " 状态 ",   dataIndex: "status",
      render: (s) => (s ? " 启用 " : " 禁用 ") },
  ];
  return <>
    <Input.Search onSearch={(k) => load(1, k)} style={{ width: 240 }} />
    <Table rowKey="id" columns={columns} dataSource={data}
           pagination={{ total, pageSize: 10, onChange: (p) => load(p) }} />
  </>;
}
